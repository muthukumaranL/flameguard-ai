# FlameGuard AI - runtime dependencies for the deployed Streamlit app.
#
# Streamlit Community Cloud installs from this file, so it stays lean: only what
# app.py imports at runtime. Analysis, export, document and test tooling lives in
# requirements-dev.txt, which scripts/setup_environment.sh (.bat) installs.
#
# torch/torchvision are pinned to the CPU builds on Linux. The default PyPI wheels
# are CUDA builds that pull in several GB of nvidia-* packages and exhaust the
# Cloud's install budget. The sys_platform marker means local Windows setups keep
# whatever CUDA wheel setup_environment installed - these lines are skipped there.

--extra-index-url https://download.pytorch.org/whl/cpu

torchvision==0.26.0+cpu ; sys_platform == "linux"

pillow>=10.0
